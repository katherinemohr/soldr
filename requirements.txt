#     uv venv --python 3.12
#     uv pip install -r requirements.txt --override requirements.txt \
#         --config-settings-package "ai-dynamo-runtime:build-args=--features ais-forward-pass"
#
# --override makes these pins win over dependencies' own: ai-dynamo pins a Linux-only aisimulate nightly. The
# ais-forward-pass feature lets the Mocker use AIC timing; Dynamo's own wheels are built with it.

# from source. That needs a Rust toolchain (https://rustup.rs) and protoc (brew install protobuf), and takes about
# 20 minutes the first time.
